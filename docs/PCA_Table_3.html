6. Análisis de componentes principales (PCA)

# PCA con variables cuantitativas estandarizadas
# accessions y nacl se incorporan como variables suplementarias
mv <- dtx_pca %>%
  PCA(
    scale.unit = TRUE,
    graph = FALSE,
    quali.sup = c(1, 2)
  )
